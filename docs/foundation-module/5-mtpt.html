§1. Main rendering recursion. Do not call this function directly: use the API in Markdown Rendering.

void MDRenderPlain::recurse(OUTPUT_STREAM, markdown_render *rdr, markdown_item *md, int mode) {
    if (rdr == NULL) internal_error("no render details in recursion");
    if (md == NULL) return;
    if (MarkdownVariations::intervene_in_rendering(rdr->variation, rdr, OUT, md, mode)) return;
    int old_mode = mode;
    switch (md->type) {
        /* Container blocks */
        case BLOCK_QUOTE_MIT:           Render a block quote1.2; break;
        case ALERT_MIT:                 Render an alert1.3; break;
        case UNORDERED_LIST_MIT:        Render a list1.4; break;
        case ORDERED_LIST_MIT:          Render a list1.4; break;
        case UNORDERED_LIST_ITEM_MIT:   Render a list item1.5; break;
        case ORDERED_LIST_ITEM_MIT:     Render a list item1.5; break;
        case FOOTNOTE_BODY_MIT:         Render a footnote body1.6; break;

        /* Leaf blocks */
        case PARAGRAPH_MIT:             Render a paragraph1.7; break;
        case THEMATIC_MIT:              Render a thematic break1.8; break;
        case HEADING_MIT:               Render a heading1.9; break;
        case CODE_BLOCK_MIT:            Render a code block1.10; break;
        case HTML_MIT:                  Render a raw HTML block1.11; break;
        case EMPTY_MIT:                 break;
        case TABLE_MIT:           Render a table1.12; break;
        case TICKBOX_MIT:               Render a tickbox1.13; break;

        /* Inline types */
        case PLAIN_MIT:                 Render plain text1.14; break;
        case LINE_BREAK_MIT:            Render a line break1.15; break;
        case SOFT_BREAK_MIT:            Render a soft line break1.16; break;
        case EMPHASIS_MIT:              Render emphasised text1.17; break;
        case STRONG_MIT:                Render strongly emphasised text1.18; break;
        case CODE_MIT:                  Render inline code sample1.19; break;
        case URI_AUTOLINK_MIT:          Render URI link1.20; break;
        case EMAIL_AUTOLINK_MIT:        Render email link1.21; break;
        case INLINE_HTML_MIT:           Render inline HTML content1.22; break;
        case LINK_MIT:                  Render link1.23; break;
        case IMAGE_MIT:                 Render image1.24; break;
        case LINK_DEST_MIT:             Render link destination1.25; break;
        case STRIKETHROUGH_MIT:         Render struck-through text1.26; break;
        case XMPP_AUTOLINK_MIT:         Render email link1.21; break;
        case TEX_MIT:                   Render maths matter in TeX1.27; break;
        case DISPLAYED_TEX_MIT:         Render maths matter in TeX1.27; break;
        case INWEB_LINK_MIT:            Render Inweb link1.28; break;

        default:                        Recurse1.1; break;
    }
    mode = old_mode;
}

§1.10. We use the convention that the first word of the info string on a fenced code block is the "language", and give it a CSS class accordingly. A piquant part of CommonMark is that the language does respect entities, but that the body of the code block does not. (It also respects backslash escapes, but we do not render the language in ESCAPES_MDRMODE mode because those have already been taken out at the parsing stage.)

Render a code block1.10 =

    mode = mode & (~ESCAPES_MDRMODE) & (~ENTITIES_MDRMODE);
    TEMPORARY_TEXT(language)
    for (int i=0; i<Str::len(md->info_string); i++) {
        inchar32_t c = Str::get_at(md->info_string, i);
        if ((c == ' ') || (c == '\t')) break;
        PUT_TO(language, c);
    }
    TEMPORARY_TEXT(language_rendered)
    if (Str::len(language) > 0) {
        md->sliced_from = language;
        md->from = 0; md->to = Str::len(language) - 1;
        if (MarkdownVariations::supports(rdr->variation, ENTITIES_MARKDOWNFEATURE))
            MDRender::slice(language_rendered, rdr, md, mode | ENTITIES_MDRMODE);
        else
            MDRender::slice(language_rendered, rdr, md, mode);
    }
    DISCARD_TEXT(language)

    MDRenderPlain::render_code_block(OUT, mode, rdr, md->stashed, language_rendered);
    DISCARD_TEXT(language_rendered)
  • This code is used in §1.

    markdown_item *alignment_markers = md->down;
    if (alignment_markers) {
        int row_count = 0;

                }

            }

§2. Character output. 

void MDRenderPlain::char(OUTPUT_STREAM, inchar32_t c, int mode) {
    if (mode & TOLOWER_MDRMODE) c = Characters::tolower(c);
    if (mode & URI_MDRMODE) {
        if (c >= 0x10000) {
            MARKDOWNPLAIN_URI_HEX(0xF0 + (unsigned char) (c >> 18));
            MARKDOWNPLAIN_URI_HEX(0x80 + (unsigned char) ((c >> 12) & 0x3f));
            MARKDOWNPLAIN_URI_HEX( 0x80 + (unsigned char) ((c >> 6) & 0x3f));
            MARKDOWNPLAIN_URI_HEX(0x80 + (unsigned char) (c & 0x3f));
        } else if (c >= 0x800) {
            MARKDOWNPLAIN_URI_HEX(0xE0 + (unsigned char) (c >> 12));
            MARKDOWNPLAIN_URI_HEX(0x80 + (unsigned char) ((c >> 6) & 0x3f));
            MARKDOWNPLAIN_URI_HEX(0x80 + (unsigned char) (c & 0x3f));
        } else if (c >= 0x80) {
            MARKDOWNPLAIN_URI_HEX(0xC0 + (unsigned char) (c >> 6));
            MARKDOWNPLAIN_URI_HEX(0x80 + (unsigned char) (c & 0x3f));
        } else {
            switch (c) {
                case '[': MARKDOWNPLAIN_URI_HEX((unsigned char) c); break;
                case '\\':MARKDOWNPLAIN_URI_HEX((unsigned char) c); break;
                case '\"':MARKDOWNPLAIN_URI_HEX((unsigned char) c); break;
                case ']': MARKDOWNPLAIN_URI_HEX((unsigned char) c); break;
                case '`': MARKDOWNPLAIN_URI_HEX((unsigned char) c); break;
                case ' ': MARKDOWNPLAIN_URI_HEX((unsigned char) c); break;
                default: PUT(c); break;
            }
        }
    } else {
        PUT(c);
    }
}
