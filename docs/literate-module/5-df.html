§2. Rendering. The weave tree will be rendered by traversing it, and rendering each node in its turn.

void DebuggingWeaving::render(weave_format *self, text_stream *OUT, heterogeneous_tree *tree) {
    weave_document_node *C = RETRIEVE_POINTER_weave_document_node(tree->root->content);
    debugging_render_state drs;
    Initialise the render state2.2;
    Trees::traverse_from(tree->root, &DebuggingWeaving::render_visit, (void *) &drs, 0);
    Append the Markdown traces2.3;
}

§3. So, then, the visiting function, called on each node. C does not allow switch statements whose cases are not literal constants, so there's a big contrived if instead. But it is morally a switch.

int DebuggingWeaving::render_visit(tree_node *N, void *state, int L) {
    debugging_render_state *drs = (debugging_render_state *) state;
    text_stream *OUT = drs->OUT;
    for (int i=0; i<L; i++) WRITE("    ");
    WRITE("%S: ", N->type->node_type_name);
    int id = N->type->allocation_id;
    if (id < 100) drs->observed[id]++;

    /* Document superstructure */

         if (N->type == weave_document_node_type) Render document3.1
    else if (N->type == weave_head_node_type) Render head3.2
    else if (N->type == weave_body_node_type) Render body3.3
    else if (N->type == weave_tail_node_type) Render tail3.4

    /* Large-scale structure */

    else if (N->type == weave_chapter_node_type) Render chapter3.5
    else if (N->type == weave_chapter_header_node_type) Render chapter header3.6
    else if (N->type == weave_chapter_footer_node_type) Render chapter footer3.7
    else if (N->type == weave_section_node_type) Render section3.8
    else if (N->type == weave_section_header_node_type) Render section header3.9
    else if (N->type == weave_section_footer_node_type) Render section footer3.10
    else if (N->type == weave_section_purpose_node_type) Render section purpose3.11
    else if (N->type == weave_toc_node_type) Render toc3.12
    else if (N->type == weave_toc_line_node_type) Render toc line3.13

    /* Small-scale structure */

    else if (N->type == weave_subheading_node_type) Render subheading3.14
    else if (N->type == weave_subsubheading_node_type) Render subsubheading3.15
    else if (N->type == weave_paragraph_heading_node_type) Render paragraph heading3.16
    else if (N->type == weave_material_node_type) Render material3.17

    /* Code-like material */

    else if (N->type == weave_holon_declaration_node_type) Render holon declaration3.18
    else if (N->type == weave_code_line_node_type) Render code line3.19
    else if (N->type == weave_holon_usage_node_type) Render holon usage3.20
    else if (N->type == weave_tangler_command_node_type) Render tangler command3.21
    else if (N->type == weave_verbatim_node_type) Render verbatim3.22
    else if (N->type == weave_source_code_node_type) Render source code3.23
    else if (N->type == weave_function_defn_node_type) Render function defn3.24
    else if (N->type == weave_function_usage_node_type) Render function usage3.25
    else if (N->type == weave_comment_in_holon_node_type) Render comment in holon3.26
    else if (N->type == weave_defn_node_type) Render defn3.27

    /* Commentary and gadget material */

    else if (N->type == weave_markdown_node_type) Render Markdown3.28
    else if (N->type == weave_audio_node_type) Render audio3.29
    else if (N->type == weave_carousel_slide_node_type) Render carousel slide3.30
    else if (N->type == weave_download_node_type) Render download3.31
    else if (N->type == weave_embed_node_type) Render embed3.32
    else if (N->type == weave_figure_node_type) Render figure3.33
    else if (N->type == weave_raw_HTML_node_type) Render raw HTML3.34
    else if (N->type == weave_video_node_type) Render video3.35

    /* Paragraph tail material */

    else if (N->type == weave_index_begins_node_type) Render index begins3.36
    else if (N->type == weave_index_lemma_node_type) Render index lemma3.37
    else if (N->type == weave_index_ends_node_type) Render index ends3.38
    else if (N->type == weave_endnote_node_type) Render endnote3.39
    else if (N->type == weave_locale_node_type) Render locale3.40
    else if (N->type == weave_endnote_text_node_type) Render endnote text3.41

    else internal_error("no HTML rendering for this node type");

    WRITE("\n");
    return TRUE;
}

§4. And the above make liberal use of these helpers:

void DebuggingWeaving::show_text(text_stream *OUT, text_stream *text, int limit) {
    WRITE(" <");
    for (int i=0; (i<limit) && (i<Str::len(text)); i++)
        if (Str::get_at(text, i) == '\n')
            WRITE("\\n");
        else
            PUT(Str::get_at(text, i));
    WRITE(">");
    if (Str::len(text) > limit) WRITE(" ... continues to %d chars", Str::len(text));
}

void DebuggingWeaving::show_para(text_stream *OUT, ls_paragraph *par) {
    WRITE(" P%S", par->paragraph_number);
    text_stream *title = LiterateSource::par_title(par);
    if (Str::len(title) > 0) WRITE("'%S'", title);
}
